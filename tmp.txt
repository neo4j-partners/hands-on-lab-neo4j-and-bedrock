  # 4. Add yourself as project member (replace USER_ID with your IAM user/role)
  # For IAM user:
  aws datazone create-project-membership \
    --domain-identifier $DOMAIN_ID \
    --project-identifier $PROJECT_ID \
    --member "userIdentifier=user/YOUR_USERNAME" \
    --designation PROJECT_OWNER \
    --region $REGION

  # For IAM role (like from CloudShell):
  aws datazone create-project-membership \
    --domain-identifier $DOMAIN_ID \
    --project-identifier $PROJECT_ID \
    --member "userIdentifier=arn:aws:iam::ACCOUNT_ID:role/ROLE_NAME" \
    --designation PROJECT_OWNER \
    --region $REGION

  # 5. List project profiles (check if any exist)
  aws datazone list-project-profiles \
    --domain-identifier $DOMAIN_ID \
    --region $REGION

  # 6. List available blueprints
  aws datazone list-blueprints \
    --domain-identifier $DOMAIN_ID \
    --region $REGION \
